import argparse
import os
import sys
import tempfile
from pathlib import Path, PurePosixPath

import boto3
from botocore.exceptions import ClientError

WILDCARD_CHARS = ("*", "?")


class ReturnCodeArgumentParser(argparse.ArgumentParser):
    """引数エラーを終了コード1で終了するパーサー。"""

    def error(self, message: str) -> None:
        self.print_usage(sys.stderr)
        self.exit(1, f"{self.prog}: 引数エラー: {message}\n")


def parse_bool(value: str) -> bool:
    """true/false文字列をboolに変換する。"""
    normalized = value.strip().lower()

    if normalized == "true":
        return True
    if normalized == "false":
        return False

    raise argparse.ArgumentTypeError(
        "上書きオプションは true または false を指定してください"
    )


def parse_args(argv=None) -> argparse.Namespace:
    parser = ReturnCodeArgumentParser(
        description="S3のファイルをEFSにコピーします"
    )

    parser.add_argument("file", help="コピー元(s3://バケット/キー)")
    parser.add_argument(
        "file_path",
        help="コピー先(EFSのファイルパス、またはディレクトリパス)",
    )
    parser.add_argument(
        "over_write_flg",
        nargs="?",
        default="true",
        type=parse_bool,
        help="上書きする場合は true、しない場合は false(省略時は true)",
    )

    return parser.parse_args(argv)


def split_s3_uri(uri: str):
    """
    s3://バケット/キー をバケットとキーに分割する。

    形式が不正な場合は ValueError を返す。
    """
    if not uri.startswith("s3://"):
        raise ValueError(f"コピー元の形式が不正です(s3://バケット/キー): {uri}")
    if any(char in uri for char in WILDCARD_CHARS):
        raise ValueError(f"コピー元にワイルドカードは指定できません: {uri}")

    bucket, _, key = uri[len("s3://"):].partition("/")
    if not bucket or not key:
        raise ValueError(f"コピー元にバケットまたはキーがありません: {uri}")
    if key.endswith("/"):
        raise ValueError(f"コピー元にディレクトリは指定できません: {uri}")

    return bucket, key


def create_s3_client():
    """
    S3クライアントを作成する。

    接続先は環境変数 S3_ENDPOINT_URL があればそちらを使う(試験環境: Floci)。
    認証情報は環境変数、AWSプロファイル、IAMロールなどから取得する。
    """
    session = boto3.Session(
        region_name=os.environ.get("AWS_DEFAULT_REGION", "ap-northeast-1"),
    )
    return session.client(
        "s3",
        endpoint_url=os.environ.get("S3_ENDPOINT_URL") or None,
    )


def run(args: argparse.Namespace) -> int:
    """
    処理結果を返す。

    0: 正常終了
    1: 処理エラー(引数不正、S3にファイルなし、上書き不可)
    -1: 例外(S3接続・権限、EFS書込、ダウンロード失敗、想定外)
    """
    try:
        if not args.file or not args.file_path:
            print("処理エラー: コピー元またはコピー先が空です", file=sys.stderr)
            return 1

        try:
            bucket, key = split_s3_uri(args.file)
        except ValueError as exc:
            print(f"処理エラー: {exc}", file=sys.stderr)
            return 1

        # コピー先がディレクトリの場合は、コピー元のファイル名でコピーする。
        filename = PurePosixPath(key).name
        destination_path = Path(args.file_path)
        if destination_path.is_dir():
            destination_path = destination_path / filename
        destination_dir = destination_path.resolve().parent

        # コピー先フォルダが存在しない場合は例外とする(作成しない)。
        if not destination_dir.is_dir():
            print(
                f"例外: コピー先ディレクトリが存在しません: {destination_dir}",
                file=sys.stderr,
            )
            return -1
        if not os.access(destination_dir, os.W_OK | os.X_OK):
            print(
                f"例外: コピー先ディレクトリに書込権限がありません: {destination_dir}",
                file=sys.stderr,
            )
            return -1

        if destination_path.exists():
            if not args.over_write_flg:
                print(
                    f"処理エラー: コピー先にファイルが存在します: {destination_path}",
                    file=sys.stderr,
                )
                return 1
            if not os.access(destination_path, os.W_OK):
                print(
                    f"例外: 既存ファイルが上書きできません: {destination_path}",
                    file=sys.stderr,
                )
                return -1

        s3 = create_s3_client()

        # コピー元の存在確認。ファイルなし、バケットなしは処理エラー、
        # 権限や接続の問題は例外(-1)とする。
        try:
            s3.head_object(Bucket=bucket, Key=key)
        except ClientError as exc:
            code = str(exc.response.get("Error", {}).get("Code", ""))
            if code in ("404", "NoSuchKey", "NotFound", "NoSuchBucket"):
                print(
                    f"処理エラー: S3にコピー元が存在しません: s3://{bucket}/{key}",
                    file=sys.stderr,
                )
                return 1
            raise

        # ダウンロード途中のファイルがコピー先に残らないよう、一時ファイルを使う。
        fd, temporary_name = tempfile.mkstemp(
            dir=destination_dir,
            prefix=f".{filename}.",
            suffix=".part",
        )
        os.close(fd)
        temporary_path = Path(temporary_name)

        try:
            s3.download_file(bucket, key, str(temporary_path))

            if args.over_write_flg:
                # 上書きあり: 一時ファイルをコピー先へ置き換える。
                os.replace(temporary_path, destination_path)
            else:
                # 上書きなし: コピー先が既に作られていた場合は上書きしない。
                try:
                    os.link(temporary_path, destination_path)
                except FileExistsError:
                    print(
                        f"処理エラー: コピー先にファイルが存在します: "
                        f"{destination_path}",
                        file=sys.stderr,
                    )
                    return 1
        finally:
            temporary_path.unlink(missing_ok=True)

        print(f"保存しました: {destination_path}")
        return 0

    except Exception as exc:
        print(f"例外: {exc}", file=sys.stderr)
        return -1


def main(argv=None) -> int:
    args = parse_args(argv)
    result = run(args)

    # OSのプロセス終了コードでは負数を移植性のある形で返せないため、
    # 例外(-1)はコマンドライン上では2に変換する。
    if result == -1:
        return 2

    return result


if __name__ == "__main__":
    raise SystemExit(main())
