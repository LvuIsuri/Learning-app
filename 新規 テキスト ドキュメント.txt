class ApExecExReqSeikyusienService(appBase.AppBaseClass):

    # 外部システム業務AP起動要求（HTTP・C++ライブラリ）（フォーム送信形式）から呼び出され、業務固有情報（フォーム）などを引数として、
    # コールバック処理を行う。

    def __init__(self, business_info_list,
                 retry_flag, retry_count, retry_wait_time
                 ):

        # ApExecExReqHttpFormServiceクラスの初期化
        # param business_info_list: 業務固有情報
        # param retry_flag: リトライ要否
        # param retry_count: リトライ回数
        # param retry_wait_time: リトライ間隔

        self.business_info_list = business_info_list
        self.retry_flag = retry_flag
        self.retry_count = retry_count
        self.retry_wait_time = retry_wait_time

        key_info = ' "' + self.business_info_list + '" '
        super().__init__(__name__, key_info)
        self.sys_err_code = 1
        self.result_err_code = -1

    def get_url(self):
        """
        # URL作成
        # HTTP通信で用いるURLを作成し、返却する。
        # :return: URL
        """
        # 接続先URL取得
        # 設定ファイル（FL_001/クライアント・サーバ共通設定ファイル．外部接続先URL）から引数．要求先をキーに接続先URLIDを検索し、IDに紐づく
        # スキーム・ホスト名を取得する。
        # 接続先Urlを取得するために必要なcategory_valueを定義する(現時点でUrlは単数、複数になった場合はサービスの引数を変えてcategory_valueを振り分ける)
        category_value='SEIKYUSIENAPI'
        scheme_host_port = confUtils.ConfUtils.get_conf_file(confUtils.XmlFiles.COMN ,category_value ,'schemeHostPort', 'externalURL')
        # URL取得に失敗した場合
        if scheme_host_port is None:
            # 監視ログにエラーログを出力する。
            # メッセージID：e.p.9009
            # 埋め込み文字{0}：設定ファイル取得キー
            self.output_log(self.location(), self.sys_err_code, 'e.p.9009', [category_value])
            # 例外を発生させる。
            raise LookupError
        # 
        # 取得したスキーム・ホスト名・ポート番号を元に接続先URL「external_url」を設定する。
        external_url = scheme_host_port
        return external_url

    def http_external_form(self, external_url, business_info_list):
        """
        HTTP通信実行
        HTTP通信を実行し、結果を返却する。
         :param external_url: 接続先URL
         :param business_info_list: 業務固有情報
         :return: HTTPステータスコード, curlコード, 業務AP終了ステータス, エラーメッセージ
         """
         
        # ポストデータの作成
        curl = pycurl.Curl()
        byte_io = io.BytesIO()
            
        # ●コールバックで使用する情報は同じ
        content_type = "Content-Type: application/x-www-form-urlencoded; charset=Windows-31J"
        # 業務固有情報を半角スペース区切りでキー、値の形式に変換する。
        # 請求区分、請求依頼番号、依頼内容区分、送信回、結果コード、エラーコードが来る想定
        splited_business_info_list = business_info_list.split(' ')
        business_info_dict = {splited_business_info_list[index] : splited_business_info_list[index + 1]
                              for index in range(0, len(splited_business_info_list), 2)}
        request_body = urlencode(business_info_dict)
        
        curl.setopt(pycurl.URL, external_url)
        curl.setopt(pycurl.CUSTOMREQUEST, 'POST')
        curl.setopt(pycurl.HTTP_VERSION, pycurl.CURL_HTTP_VERSION_1_1)
        curl.setopt(pycurl.WRITEFUNCTION, byte_io.write)
        
        # ●環境変数API-KEY取得
        api_key = os.getenv('API_KEY')
        # 取得したAPIKEY情報をヘッダに設定
        curl.setopt(pycurl.HTTPHEADER, [content_type , 'apikey:' + api_key])
        curl.setopt(pycurl.POSTFIELDS, request_body)

        # HTTPリクエスト開始
        try:
            curl.perform()
            curl_code = pycurl.E_OK

            # HTTPステータスコード取得
            http_status_code = curl.getinfo(pycurl.HTTP_CODE)

        except pycurl.error as e:
            curl_code = e.args[0]

            # HTTP通信自体が実行できない場合はhttp_status_codeに「-」（半角ハイフン）を設定する。
            http_status_code = '-'

        result = 0
        err_msg = None

        if (http_status_code == 200):
            # レスポンスは半角文字のみで返却される
            res = byte_io.getvalue().decode('cp932').strip()
            result = int(res)

        return http_status_code, curl_code, result, err_msg

    def do_http_external_form(self):
        # HTTP通信
        # HTTP通信メソッドを用いて、リトライを含めたHTTP通信を実施する。
        # :return 処理結果、処理結果（外部システム）

        max_retry = 0

        # URLを取得する。
        try:
            external_url = self.get_url()
        except LookupError:
            # 異常終了する。
            # 処理結果に'1'（システムエラー）を設定し返却する。
            return self.sys_err_code

        # 引数．リトライ要否が要の場合
        if self.retry_flag == "true":
            # 引数．リトライ回数が設定されている場合
            if self.retry_count != "0":
                # 引数．リトライ回数をリトライ回数に設定する。
                max_retry = int(self.retry_count)
            # 上記以外の場合
            else:
                # 設定ファイル（FL_002/サーバサイド設定ファイル．リトライ．リトライ回数）をリトライ回数に設定する。
                max_retry = int(confUtils.ConfUtils.get_conf_file(confUtils.XmlFiles.COMN, 'retry', 'retryCount'))

            # 引数．リトライ間隔が設定されている場合
            if self.retry_wait_time != "0":
                # 引数．リトライ間隔をリトライ待機時間に設定する。
                retry_wait_time = int(self.retry_wait_time)
            # 上記以外の場合
            else:
                # 設定ファイル（FL_002/サーバサイド設定ファイル．リトライ．リトライ待機時間）を
                # リトライ待機時間に設定する。
                retry_wait_time = int(
                    confUtils.ConfUtils.get_conf_file(confUtils.XmlFiles.COMN, 'retry', 'retryWaitTime'))
        # 上記以外の場合
        # 0をリトライ回数に設定する。（初期値が0なので処理は不要）

        # リトライカウンター初期化
        retry_counter = 0

        while True:
            http_status_code, curl_code, result, err_msg = self.http_external_form(external_url,
                                                                                   self.business_info_list)
            # 通信異常判定
            # HTTPステータスコードが200以外
            if http_status_code == 200:
                break

            # リトライ判定
            # 通信異常（pycurlエラーコードが'7'（ホスト接続失敗）、'28'（タイムアウト）、HTTPステータスコードが500または503）
            # かつ（設定ファイル（FL_002/サーバサイド設定ファイル．リトライ．リトライ回数）以下の場合）
            if ((curl_code == pycurl.E_COULDNT_CONNECT or curl_code == pycurl.E_OPERATION_TIMEDOUT
                 or http_status_code == 500 or http_status_code == 503) and retry_counter < max_retry):

                # 監視ログにワーニングログを出力する。
                # メッセージID：w.p.9001
                # 埋め込み文字{0}：HTTPステータスコード
                # 埋め込み文字{1}：URL
                # 但し、HTTP通信自体が実行できない場合は{0}に「-」（半角ハイフン）を設定する。
                self.output_log(self.location(), '', 'w.p.9001', [http_status_code, external_url])
                # 設定したリトライ待機時間だけ処理を待つ。
                time.sleep(retry_wait_time)

                # HTTP通信をリトライする。
                # 無限ループに入っているため、breakしなければリトライし続ける。
                # リトライカウンター加算
                retry_counter += 1

            # 上記以外の場合
            else:
                # 監視ログにエラーログを出力する。
                # メッセージID：e.p.9010
                # 埋め込み文字{0}：HTTPステータスコード
                # 埋め込み文字{1}：err_msg
                # 但し、HTTP通信自体が実行できない場合は{0}に「-」（半角ハイフン）を設定する。
                self.output_log(self.location(), self.sys_err_code, 'e.p.9010',
                                [f'{http_status_code}, {err_msg}', external_url])

                # 異常終了する。
                # 処理結果に"1"（外部システムAPが異常終了またはシステムエラー）、
                # 処理結果（外部システム）に"-1"（異常終了）を設定し返却する。
                return self.sys_err_code, self.result_err_code

        # 結果返却
        # 業務AP終了ステータス戻り値判定
        # 業務AP終了ステータスが”0”以外（外部システムAPが異常終了時のステータスコード、システムエラー）の場合
        if result != 0:
            # 異常終了する。
            # 処理結果に"1"（外部システムAPが異常終了）、処理結果（外部システム）に業務AP終了ステータスを設定し返却する。
            return self.sys_err_code, result

        # 上記以外の場合
        # 正常終了する。
        # 処理結果に"0"（正常終了）、処理結果（外部システム）に"0"（正常終了）を設定し返却する。

        return 0, 0

        """
        frame = inspect.currentframe().f_back
        return frame.f_lineno

class ExclusiveFileHandler(logging.handlers.TimedRotatingFileHandler):
    def __init__(self, file_name, key_info=None):
        """
        初期化
        ExclusiveFileHandlerクラスの初期化を実行する。
        :param file_name:ファイル名
        """
        # ログファイルのローテートを1日単位に設定
        super(ExclusiveFileHandler, self).__init__(filename=file_name, when='midnight', delay=True)
        if key_info:
            self.lock_file_path = FilePath.get_sqs_temp_dir() + "lockFile-" + key_info
        else:
            self.lock_file_path = FilePath.get_lock_file_path()

    def shouldRollover(self, record):
        """
        ログローテート判定
        ログローテート可否をファイルのタイプスタンプを用いて判定する。
        :param record:書き込みの際に用いられるパラメタ
        """
        filename = self.baseFilename
        if os.path.exists(filename):
            t = os.stat(filename)[ST_MTIME]
        else:
            t = int(time.time())
        self.rolloverAt = self.computeRollover(t)
        return super().shouldRollover(record)

    def emit(self, record):
        """
        マルチプロセス排他ログ（排他処理ハンドラー）
        マルチプロセスでログを書き込む際に排他制御を行いながら1ファイルにログを書き込む。
        :param record:書き込みの際に用いられるパラメタ
        """

        # ファイルロック処理
        #  設定ファイル（FL_005/排他ロックファイル）に対してファイルロックされているかを確認する。
        with open(self.lock_file_path, 'w') as oLockFile:
            oLockFile.write('')
            #  ファイルロックされていない場合
            #  flock関数で設定ファイル（FL_005/排他ロックファイル）をロック状態に更新する。
            #  ファイルロックされている場合
            #  ファイルロックが解除されるまで待機する。
            # ファイルロックが解除されると自動的に1.1-aの処理を行う。
            fcntl.flock(oLockFile.fileno(), fcntl.LOCK_EX)

            try:
                #  ログ出力処理
                #  ログファイルを開く。
                #  ログファイルにログ出力する。
                super().emit(record)

            finally:
                #  ログファイルを閉じる。
                super().close()

                #  ロック解除処理
                #  ファイルロックを解除する。
                fcntl.flock(oLockFile.fileno(), fcntl.LOCK_UN)


class LogUtils:

    # エラーレベル変換メソッド
    CRITICAL = 50
    FATAL = CRITICAL
    ERROR = 40
    WARNING = 30
    WARN = WARNING
    INFO = 20
    DEBUG = 10
    NOTSET = 0

    _nameToLevel = {
        'CRITICAL': CRITICAL,
        'FATAL': FATAL,
        'ERROR': ERROR,
        'WARN': WARNING,
        'WARNING': WARNING,
        'INFO': INFO,
        'DEBUG': DEBUG,
        'NOTSET': NOTSET,
    }

    __log_format = 'date:%(asctime)s.%(msecs)03d\tlevel:%(levelname)s logger:{0}'\
                   + '\terrorPoint:%(line_number)d' + '\terrorData:{1}'\
                   + '\terrorCode:%(error_code)s\tmessage:[%(message_id)s]%(message)s'
    __log_date = '%Y-%m-%d %H:%M:%S'

    @classmethod
    def _check_level(cls, level):
        """
        エラーレベル変換
        INFO、WARNING、ERRORを数値に書き換えて返却する。
        :param level:エラーレベル（文字列）
        :return:エラーレベル（数値）
        """

        if isinstance(level, int):
            rv = level
        elif str(level) == level:
            if level not in cls._nameToLevel:
                raise ValueError("Unknown level: %r" % level)
            rv = cls._nameToLevel[level]
        else:
            raise TypeError("Level not an integer or a valid string: %r" % level)
        return rv

    @classmethod
    def get_logger(cls, name, key_info):
        """
        ロガー生成
        ログのフォーマットを設定し、ロガーを作成する。
        :param name: ロガーの名前
        :param key_info: ログ書き出し元が判別できるような情報
        :return logger: ロガー
        """
        logger = logging.getLogger(name)
        logger.setLevel(logging.INFO)

        #  ログフォーマットを設定する。
        formatter = logging.Formatter(
            fmt=cls.__log_format.format(name, key_info),
            datefmt=cls.__log_date)

        #  排他処理ハンドラーインスタンスをコンストラクタ引数にログファイル名を渡して生成する。
        # ログファイル名についてはログ設計書（監視ログ_Python）を参照のこと。
        # ログファイルのパス
        file_path = FilePath.get_log_file_path()
        exclusive_handler = ExclusiveFileHandler(file_path)
        exclusive_handler.setFormatter(formatter)
        logger.addHandler(exclusive_handler)

        return logger

    @classmethod
    def get_biz_logger(cls, name, key_info):
        """
        ロガー生成
        ログのフォーマットを設定し、ロガーを作成する。
        :param name: ロガーの名前
        :param key_info: ログ書き出し元が判別できるような情報
        :return logger: ロガー
        """
        logger = logging.getLogger(name + "_" + key_info)
        logger.setLevel(logging.INFO)

        #  ログフォーマットを設定する。
        formatter = logging.Formatter(
            fmt=cls.__log_format.format(name, key_info),
            datefmt=cls.__log_date)

        #  排他処理ハンドラーインスタンスをコンストラクタ引数にログファイル名を渡して生成する。
        # ログファイル名についてはログ設計書（監視ログ_Python）を参照のこと。
        # ログファイルのパス
        file_path = FilePath.get_log_path() + "kikin-application-ctrlqueuercv-" + key_info + "-apbase-pymodule.log"
        exclusive_handler = ExclusiveFileHandler(file_path, key_info)
        exclusive_handler.setFormatter(formatter)
        logger.addHandler(exclusive_handler)

        return logger

    @classmethod
    def get_event_logger(cls, name, key_info):
        """
        ロガー生成
        ログのフォーマットを設定し、ロガーを作成する。
        :param name: ロガーの名前
        :param key_info: ログ書き出し元が判別できるような情報
        :return logger: ロガー
        """
        logger = logging.getLogger(name + "_" + key_info)
        logger.setLevel(logging.INFO)

        #  ログフォーマットを設定する。
        formatter = logging.Formatter(
            fmt=cls.__log_format.format(name, key_info),
            datefmt=cls.__log_date)

        #  Windowsイベントログハンドラをコンストラクタ引数にサービス名を渡して生成する。
        ntl = logging.handlers.NTEventLogHandler(name)
        ntl.setFormatter(formatter)
        logger.addHandler(ntl)

        return logger

    @classmethod
    def output_log(cls, logger, line_number, error_code, message_id, inner_message=''):
        """
        ログ書き出し実行
        ログ書き出しを実行する。
        :param logger:ロガー
        :param line_number: 呼び出し元の行数
        :param error_code:ログに書き出すエラーコード
        :param message_id:ログに書き出すメッセージのID
        :param inner_message:ログへの埋め込み文字
        """
        #  ログ出力
        #  ログメッセージ取得
        #  設定ファイル（FL_004/エラーメッセージ設定ファイル）からエラーメッセージIDをキーにエラーレベル、エラーメッセージを取得する。
        # 共通部品設計書（PY_001/設定ファイル取得）を参照のこと。

        # ログファイル書き出し用のエラー情報を設定ファイルから取得する。
        try:
            err_level = confUtils.ConfUtils.get_conf_file(confUtils.XmlFiles.ERR, message_id, 'errLevel')
            err_message = confUtils.ConfUtils.get_conf_file(confUtils.XmlFiles.ERR, message_id, 'errMsg')

        except:
            log_dict = {'line_number': line_number, 'error_code': error_code,
                        'message_id': 'e.p.9001'}
            logger.log(cls._check_level('ERROR'), 'システムエラーが発生しました。'
                                                  'ログメッセージ設定ファイルの読み込みに失敗しました。'
                       , extra=log_dict)
            return

        # inner_messageを配列で受け取って埋め込み
        for index in range(len(inner_message)):
            err_message = err_message.replace('{' + str(index) + '}', str(inner_message[index]))

        #  ログ出力
        #  初期化処理で生成したロガーを取得する。

        log_dict = {'line_number': line_number, 'error_code': error_code,
                    'message_id': message_id}

        logger.log(cls._check_level(str(err_level).upper()), err_message, extra=log_dict)
