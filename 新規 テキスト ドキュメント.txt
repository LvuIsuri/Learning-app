class ExclusiveFileHandler(logging.handlers.TimedRotatingFileHandler):
    def __init__(self, file_name, key_info=None):
        """
        初期化
        ExclusiveFileHandlerクラスの初期化を実行する。
        :param file_name:ファイル名
        """
        # ログファイルのローテートを1日単位に設定
        super(ExclusiveFileHandler, self).__init__(filename=file_name, when='midnight', delay=True)
        if key_info:
            self.lock_file_path = FilePath.get_sqs_temp_dir() + "lockFile-" + key_info
        else:
            self.lock_file_path = FilePath.get_lock_file_path()

    def shouldRollover(self, record):
        """
        ログローテート判定
        ログローテート可否をファイルのタイプスタンプを用いて判定する。
        :param record:書き込みの際に用いられるパラメタ
        """
        filename = self.baseFilename
        if os.path.exists(filename):
            t = os.stat(filename)[ST_MTIME]
        else:
            t = int(time.time())
        self.rolloverAt = self.computeRollover(t)
        return super().shouldRollover(record)

    def emit(self, record):
        """
        マルチプロセス排他ログ（排他処理ハンドラー）
        マルチプロセスでログを書き込む際に排他制御を行いながら1ファイルにログを書き込む。
        :param record:書き込みの際に用いられるパラメタ
        """

        # ファイルロック処理
        #  設定ファイル（FL_005/排他ロックファイル）に対してファイルロックされているかを確認する。
        with open(self.lock_file_path, 'w') as oLockFile:
            oLockFile.write('')
            #  ファイルロックされていない場合
            #  flock関数で設定ファイル（FL_005/排他ロックファイル）をロック状態に更新する。
            #  ファイルロックされている場合
            #  ファイルロックが解除されるまで待機する。
            # ファイルロックが解除されると自動的に1.1-aの処理を行う。
            fcntl.flock(oLockFile.fileno(), fcntl.LOCK_EX)

            try:
                #  ログ出力処理
                #  ログファイルを開く。
                #  ログファイルにログ出力する。
                super().emit(record)

            finally:
                #  ログファイルを閉じる。
                super().close()

                #  ロック解除処理
                #  ファイルロックを解除する。
                fcntl.flock(oLockFile.fileno(), fcntl.LOCK_UN)


class LogUtils:

    # エラーレベル変換メソッド
    CRITICAL = 50
    FATAL = CRITICAL
    ERROR = 40
    WARNING = 30
    WARN = WARNING
    INFO = 20
    DEBUG = 10
    NOTSET = 0

    _nameToLevel = {
        'CRITICAL': CRITICAL,
        'FATAL': FATAL,
        'ERROR': ERROR,
        'WARN': WARNING,
        'WARNING': WARNING,
        'INFO': INFO,
        'DEBUG': DEBUG,
        'NOTSET': NOTSET,
    }

    __log_format = 'date:%(asctime)s.%(msecs)03d\tlevel:%(levelname)s logger:{0}'\
                   + '\terrorPoint:%(line_number)d' + '\terrorData:{1}'\
                   + '\terrorCode:%(error_code)s\tmessage:[%(message_id)s]%(message)s'
    __log_date = '%Y-%m-%d %H:%M:%S'

    @classmethod
    def _check_level(cls, level):
        """
        エラーレベル変換
        INFO、WARNING、ERRORを数値に書き換えて返却する。
        :param level:エラーレベル（文字列）
        :return:エラーレベル（数値）
        """

        if isinstance(level, int):
            rv = level
        elif str(level) == level:
            if level not in cls._nameToLevel:
                raise ValueError("Unknown level: %r" % level)
            rv = cls._nameToLevel[level]
        else:
            raise TypeError("Level not an integer or a valid string: %r" % level)
        return rv

    @classmethod
    def get_logger(cls, name, key_info):
        """
        ロガー生成
        ログのフォーマットを設定し、ロガーを作成する。
        :param name: ロガーの名前
        :param key_info: ログ書き出し元が判別できるような情報
        :return logger: ロガー
        """
        logger = logging.getLogger(name)
        logger.setLevel(logging.INFO)

        #  ログフォーマットを設定する。
        formatter = logging.Formatter(
            fmt=cls.__log_format.format(name, key_info),
            datefmt=cls.__log_date)

        #  排他処理ハンドラーインスタンスをコンストラクタ引数にログファイル名を渡して生成する。
        # ログファイル名についてはログ設計書（監視ログ_Python）を参照のこと。
        # ログファイルのパス
        file_path = FilePath.get_log_file_path()
        exclusive_handler = ExclusiveFileHandler(file_path)
        exclusive_handler.setFormatter(formatter)
        logger.addHandler(exclusive_handler)

        return logger

    @classmethod
    def get_biz_logger(cls, name, key_info):
        """
        ロガー生成
        ログのフォーマットを設定し、ロガーを作成する。
        :param name: ロガーの名前
        :param key_info: ログ書き出し元が判別できるような情報
        :return logger: ロガー
        """
        logger = logging.getLogger(name + "_" + key_info)
        logger.setLevel(logging.INFO)

        #  ログフォーマットを設定する。
        formatter = logging.Formatter(
            fmt=cls.__log_format.format(name, key_info),
            datefmt=cls.__log_date)

        #  排他処理ハンドラーインスタンスをコンストラクタ引数にログファイル名を渡して生成する。
        # ログファイル名についてはログ設計書（監視ログ_Python）を参照のこと。
        # ログファイルのパス
        file_path = FilePath.get_log_path() + "kikin-application-ctrlqueuercv-" + key_info + "-apbase-pymodule.log"
        exclusive_handler = ExclusiveFileHandler(file_path, key_info)
        exclusive_handler.setFormatter(formatter)
        logger.addHandler(exclusive_handler)

        return logger

    @classmethod
    def get_event_logger(cls, name, key_info):
        """
        ロガー生成
        ログのフォーマットを設定し、ロガーを作成する。
        :param name: ロガーの名前
        :param key_info: ログ書き出し元が判別できるような情報
        :return logger: ロガー
        """
        logger = logging.getLogger(name + "_" + key_info)
        logger.setLevel(logging.INFO)

        #  ログフォーマットを設定する。
        formatter = logging.Formatter(
            fmt=cls.__log_format.format(name, key_info),
            datefmt=cls.__log_date)

        #  Windowsイベントログハンドラをコンストラクタ引数にサービス名を渡して生成する。
        ntl = logging.handlers.NTEventLogHandler(name)
        ntl.setFormatter(formatter)
        logger.addHandler(ntl)

        return logger

    @classmethod
    def output_log(cls, logger, line_number, error_code, message_id, inner_message=''):
        """
        ログ書き出し実行
        ログ書き出しを実行する。
        :param logger:ロガー
        :param line_number: 呼び出し元の行数
        :param error_code:ログに書き出すエラーコード
        :param message_id:ログに書き出すメッセージのID
        :param inner_message:ログへの埋め込み文字
        """
        #  ログ出力
        #  ログメッセージ取得
        #  設定ファイル（FL_004/エラーメッセージ設定ファイル）からエラーメッセージIDをキーにエラーレベル、エラーメッセージを取得する。
        # 共通部品設計書（PY_001/設定ファイル取得）を参照のこと。

        # ログファイル書き出し用のエラー情報を設定ファイルから取得する。
        try:
            err_level = confUtils.ConfUtils.get_conf_file(confUtils.XmlFiles.ERR, message_id, 'errLevel')
            err_message = confUtils.ConfUtils.get_conf_file(confUtils.XmlFiles.ERR, message_id, 'errMsg')

        except:
            log_dict = {'line_number': line_number, 'error_code': error_code,
                        'message_id': 'e.p.9001'}
            logger.log(cls._check_level('ERROR'), 'システムエラーが発生しました。'
                                                  'ログメッセージ設定ファイルの読み込みに失敗しました。'
                       , extra=log_dict)
            return

        # inner_messageを配列で受け取って埋め込み
        for index in range(len(inner_message)):
            err_message = err_message.replace('{' + str(index) + '}', str(inner_message[index]))

        #  ログ出力
        #  初期化処理で生成したロガーを取得する。

        log_dict = {'line_number': line_number, 'error_code': error_code,
                    'message_id': message_id}

        logger.log(cls._check_level(str(err_level).upper()), err_message, extra=log_dict)
import logging
import sys
import xml.etree.ElementTree as ET
from pathlib import Path


class _ConsoleHandler(logging.StreamHandler):
    def emit(self, record):
        self.stream = sys.stderr
        super().emit(record)


class FileCopyS3EfsBase:
    """Provide XML-catalog logging for the standalone S3/EFS copy service."""

    _LEVELS = {
        "debug": logging.DEBUG,
        "info": logging.INFO,
        "warn": logging.WARNING,
        "warning": logging.WARNING,
        "error": logging.ERROR,
        "critical": logging.CRITICAL,
        "fatal": logging.CRITICAL,
    }
    _MESSAGE_FILE = Path(__file__).with_name("fileCopyS3EfsErrMsg.xml")

    def __init__(self, logger_name: str, key_info: str):
        self.logger = logging.getLogger(logger_name)
        self.logger.setLevel(logging.INFO)
        self.logger.propagate = False
        self.key_info = key_info
        self._messages = None

    def configure_logger(self) -> None:
        """Load the adjacent XML catalog and configure console output."""
        if self._messages is not None:
            return

        if not any(isinstance(handler, _ConsoleHandler) for handler in self.logger.handlers):
            handler = _ConsoleHandler()
            handler.setFormatter(logging.Formatter(
                fmt=(
                    "date:%(asctime)s.%(msecs)03d\tlevel:%(levelname)s "
                    "logger:%(name)s\terrorPoint:%(line_number)d"
                    "\terrorData:%(key_info)s\terrorCode:%(error_code)s"
                    "\tmessage:[%(message_id)s]%(message)s"
                ),
                datefmt="%Y-%m-%d %H:%M:%S",
            ))
            self.logger.addHandler(handler)

        root = ET.parse(self._MESSAGE_FILE).getroot()
        messages = {}
        for err_info in root.findall("errInfo"):
            message_id = err_info.get("id")
            level = (err_info.findtext("errLevel") or "").strip().lower()
            message = err_info.findtext("errMsg")
            if not message_id or level not in self._LEVELS or message is None:
                raise ValueError(f"Invalid error-message entry in {self._MESSAGE_FILE}")
            messages[message_id] = (self._LEVELS[level], message)
        self._messages = messages

    def _emit_log(
        self,
        message_id: str,
        inner_message,
        line_number: int,
        error_code: int | str,
    ) -> None:
        try:
            level, message = self._messages[message_id]
        except KeyError as exc:
            raise KeyError(f"Message ID not found in {self._MESSAGE_FILE}: {message_id}") from exc

        for index, value in enumerate(inner_message):
            message = message.replace("{" + str(index) + "}", str(value))

        self.logger.log(
            level,
            message,
            extra={
                "line_number": line_number,
                "key_info": self.key_info,
                "error_code": error_code,
                "message_id": message_id,
            },
        )

    def output_log(self, message_id: str, *inner_message: str, error_code: int | str | None = None) -> None:
        """Format and emit a message selected by ID from the XML catalog."""
        line_number = sys._getframe(1).f_lineno
        self.configure_logger()
        level = self._messages[message_id][0]
        if error_code is None:
            error_code = 0 if level < logging.WARNING else 1
        self._emit_log(message_id, inner_message, line_number, error_code)

    def output_system_error(self, message: str) -> None:
        """Emit e.p.9001 from XML or a direct diagnostic if XML setup failed."""
        line_number = sys._getframe(1).f_lineno
        try:
            self.configure_logger()
            self._emit_log("e.p.9001", (message,), line_number, -1)
        except (OSError, ET.ParseError, ValueError, KeyError):
            self.logger.error(
                "システムエラーが発生しました。%s",
                message,
                extra={
                    "line_number": line_number,
                    "key_info": self.key_info,
                    "error_code": -1,
                    "message_id": "e.p.9001",
                },
            )
