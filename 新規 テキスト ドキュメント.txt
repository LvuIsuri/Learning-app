import shutil
from pathlib import Path
from unittest.mock import Mock

import pytest
from botocore.exceptions import ClientError

from kikin.fw.infra.external import fileCopyS3EfsService as service

# ---- 入力値に使う値(Excelの表記と同じ) ----
S3_SAMPLE = "s3://test-bucket/test/sample.csv"     # S3のコピー元ファイル
S3_UP = "s3://test-bucket/test/up.csv"             # S3のコピー先ファイル
S3_BUCKET_ONLY = "s3://test-bucket"                # バケット名のみ
EFS_OUT = Path("/home/bttranvu/efs_test/out")      # Excelの「/efs_test/out」に相当
S3_DATA = b"S3DATA"                                # S3側のファイル内容


@pytest.fixture(autouse=True)
def setup(monkeypatch):
    """各テストの事前条件。(@BeforeEach相当)
    EFS：/efs_test/out を空の状態にする。
    S3 ：test/sample.csv が存在する状態のMockにする。(状態を変えたい試験は、各試験内で上書きする)
    """
    # EFS
    if EFS_OUT.exists():
        shutil.rmtree(EFS_OUT)
    EFS_OUT.mkdir(parents=True)

    # S3(Mock)
    client = Mock()
    client.head_object.return_value = {}                         # オブジェクトが存在する
    client.download_file.side_effect = (                         # ダウンロード＝EFSへS3の内容を書く
        lambda bucket, key, dest: Path(dest).write_bytes(S3_DATA))
    monkeypatch.setattr(service, "create_s3_client", lambda: client)
    return client


def run(*args):
    """処理を実行し、(戻り値, ログ出力のMock)を返す。"""
    copier = service.ApCopyFileS3EfsService.__new__(service.ApCopyFileS3EfsService)
    copier.argv = list(args)
    copier.request = " ".join(args)
    copier.output_log = Mock()
    return copier.do_copy(), copier.output_log
def test1():
    """
    test1：引数が指定されていない場合
    入力値：引数なし
    """
    result, log = run()
    assert result == 1
    check_log(log, 1, "e.p.9013", "引数は2個または3個指定してください。")


def test2():
    """
    test2：コピー元ファイルが空文字の場合
    入力値：(空文字) /efs_test/out/new.csv
    """
    result, log = run("", str(EFS_OUT / "new.csv"))
    assert result == 1
    check_log(log, 1, "e.p.9013", "コピー元またはコピー先が空です。")


def test19():
    """
    test19：コピー先パスに「..」が含まれる場合
    入力値：s3://test-bucket/test/sample.csv /efs_test/out/../new.csv
    """
    result, log = run(S3_SAMPLE, str(EFS_OUT) + "/../new.csv")
    assert result == 1
    assert not (EFS_OUT.parent / "new.csv").exists(), "コピー先にファイルが作成されない想定"
    check_log(log, 1, "e.p.9013", "コピー先パスに「..」は指定できません。")


def test23(setup):
    """
    test23：コピー元ファイルがS3に存在しない場合
    入力値：s3://test-bucket/test/none.csv /efs_test/out/new.csv false
    """
    # 事前条件の変更：S3にファイルが存在しない(404)
    setup.head_object.side_effect = ClientError(
        {"Error": {"Code": "404", "Message": "not found"}}, "HeadObject")

    result, log = run("s3://test-bucket/test/none.csv", str(EFS_OUT / "new.csv"), "false")
    assert result == 1
    assert not (EFS_OUT / "new.csv").exists(), "コピー先にファイルが作成されない想定"
    check_log(log, 1, "e.p.9013", "コピー元オブジェクトが存在しません。")


def test33():
    """
    test33：コピー元にS3ファイル、コピー先にEFSファイルを指定、コピー先が新規の場合
    入力値：s3://test-bucket/test/sample.csv /efs_test/out/new.csv
    """
    result, log = run(S3_SAMPLE, str(EFS_OUT / "new.csv"))
    assert result == 0
    assert (EFS_OUT / "new.csv").read_bytes() == S3_DATA, "ファイル内容がS3と一致する想定"
    log.assert_not_called()

def check_log(log, error_code, message_id, text):
    """ログが1件出力され、エラーコード・ID・エラー内容が期待どおりであることを確認する。"""
    log.assert_called_once()
    assert log.call_args.args[1:3] == (error_code, message_id)
    assert text in log.call_args.args[3][0]
