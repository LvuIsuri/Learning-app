import logging
import os
import sys
import tempfile
from pathlib import Path, PurePosixPath

import boto3
from boto3.exceptions import S3UploadFailedError
from botocore.exceptions import ClientError

logger = logging.getLogger(__name__)

S3_PREFIX = "s3://"
WILDCARD_CHARS = ("*", "?")
# HeadObjectで「オブジェクトまたはバケットなし」と判定するエラーコード。
NOT_FOUND_CODES = ("404", "NoSuchKey", "NotFound", "NoSuchBucket")


def split_s3_uri(
    uri: str,
    label: str = "source",
    allow_empty_key: bool = False,
):
    """S3 URIからバケット名とオブジェクトキーを取得する。

    Args:
        uri: s3://バケット/キー 形式のURI。
        label: エラーメッセージに表示する識別名（例: source、destination）。
        allow_empty_key: Trueの場合、オブジェクトキーが空でも許可する。

    Returns:
        (バケット名, オブジェクトキー)。

    Raises:
        ValueError: バケット名が空、または許可されていないキーが空の場合。
    """

    # 最初の「/」より前をバケット名、後ろをオブジェクトキーとして扱う。
    bucket, _, key = uri[len(S3_PREFIX):].partition("/")

    if not bucket or (not key and not allow_empty_key):
        raise ValueError(f"{label}にバケットまたはキーがありません: {uri}")

    return bucket, key


def parse_over_write_flg(value: str) -> bool:
    """コマンドラインの上書き指定をbool値に変換する。

    Raises:
        ValueError: true/false以外（空文字を含む）が指定された場合。
    """
    # 大文字・小文字と前後の空白を許容してから値を判定する。
    normalized = value.strip().lower()

    if normalized == "true":
        return True
    if normalized == "false":
        return False

    raise ValueError("上書きオプションは true または false を指定してください")


def has_parent_reference(path: str) -> bool:
    """パスを[/]で分割し、「..」が含まれるか判定する。"""
    return ".." in path.split("/")


def create_s3_client():
    """環境変数の接続先・リージョンを使ってS3クライアントを作成する。

    S3_ENDPOINT_URLが設定されていればFlociなどの指定先を使い、
    未設定の場合はboto3の標準AWS認証情報チェーンを利用する。
    """
    session = boto3.Session(
        region_name=os.environ.get("AWS_DEFAULT_REGION", "ap-northeast-1"),
    )
    return session.client(
        "s3",
        endpoint_url=os.environ.get("S3_ENDPOINT_URL") or None,
    )


def s3_exists(s3, bucket: str, key: str) -> bool:
    """HeadObjectでS3オブジェクトの存在を確認する。

    オブジェクトまたはバケットがない場合はFalseを返す。
    AccessDeniedや接続エラーなど、存在確認以外の失敗は呼び出し元へ送出する。
    """
    try:
        s3.head_object(Bucket=bucket, Key=key)
    except ClientError as exc:
        code = str(exc.response.get("Error", {}).get("Code", ""))
        if code in NOT_FOUND_CODES:
            return False
        raise
    return True


def copy_s3_to_efs(file: str, file_path: str, over_write: bool) -> int:
    """S3からEFSへ単一ファイルをコピーする。

    戻り値:
        0: コピー成功。
        1: URI不正、コピー元なし、上書き不可などの入力・処理エラー。
        -1: EFS書込、S3接続、権限、ダウンロードなどの例外。
    """
    try:
        # コピー元URIを検証し、S3 APIで使うバケット名とキーを取得する。
        bucket, key = split_s3_uri(file, "source")
        if key.endswith("/"):
            raise ValueError(f"にディレクトリは指定できません: {file}")
    except ValueError:
        logger.warning("S3からEFSへのコピーを中止しました。コピー元URIが不正です: %s", file)
        return 1

    if has_parent_reference(file_path):
        logger.warning("S3からEFSへのコピーを中止しました。コピー先パスに「..」は指定できません: %s", file_path)
        return 1

    # ディレクトリ指定ならS3キーの末尾をファイル名としてコピー先に追加する。
    filename = PurePosixPath(key).name
    destination_path = Path(file_path)

    if destination_path.is_dir():
        destination_path = destination_path / filename
    destination_dir = destination_path.resolve().parent

    # 出力先ディレクトリは自動作成せず、存在と書込権限を事前確認する。
    if not destination_dir.is_dir():
        logger.warning("S3からEFSへのコピーを中止しました。コピー先ディレクトリが存在しません: %s", destination_dir)
        return 1

    if not os.access(destination_dir, os.W_OK | os.X_OK):
        logger.error("S3からEFSへのコピーに失敗しました。コピー先ディレクトリに書き込めません: %s", destination_dir)
        return -1

    if destination_path.is_file():
        if not over_write:
            logger.warning("S3からEFSへのコピーを中止しました。コピー先ファイルが存在し、上書きが許可されていません: %s", destination_path)
            return 1
        if not os.access(destination_path, os.W_OK):
            logger.error("S3からEFSへのコピーに失敗しました。コピー先ファイルに書き込めません: %s", destination_path)
            return -1

    s3 = create_s3_client()

    # ダウンロード前にS3上のコピー元を確認する。未存在は処理エラーとして返す。
    if not s3_exists(s3, bucket, key):
        logger.warning("S3からEFSへのコピーを中止しました。コピー元オブジェクトが存在しません: s3://%s/%s", bucket, key)
        return 1

    # 同じディレクトリに一時ファイルを作り、不完全なコピー先が残るのを防ぐ。
    fd, temporary_name = tempfile.mkstemp(
        dir=destination_dir,
        prefix=f".{filename}.",
        suffix=".part",
    )
    os.close(fd)
    temporary_path = Path(temporary_name)

    try:
        # まず一時ファイルへダウンロードし、完了後にコピー先へ反映する。
        s3.download_file(bucket, key, str(temporary_path))

        # 一時ファイルは権限0600で作成されるため、コピー先の権限を0664にする。
        # (新規・上書きともに0664。上書きの場合も既存ファイルの権限は引き継がない)
        os.chmod(temporary_path, 0o664)

        if over_write:
            # 上書きあり: 完成した一時ファイルで既存ファイルを置き換える。
            os.replace(temporary_path, destination_path)
        else:
            # 上書きなし: 競合で先にファイルが作られた場合も上書きしない。
            try:
                os.link(temporary_path, destination_path)
            except FileExistsError:
                logger.warning("S3からEFSへのコピーを中止しました。処理中に別プロセスがコピー先ファイルを作成しました: %s", destination_path)
                return 1
    finally:
        # 成功・失敗のどちらでも一時ファイルを残さない。
        temporary_path.unlink(missing_ok=True)

    return 0


def copy_efs_to_s3(file: str, file_path: str, over_write: bool) -> int:
    """EFSからS3へ単一ファイルをコピーする。

    戻り値:
        0: アップロード成功。
        1: コピー元なし、URI不正、上書き不可、コピー先バケットなしなど。
        -1: S3接続、権限、その他アップロード時の例外。
    """

    if has_parent_reference(file):
        logger.warning("EFSからS3へのコピーを中止しました。コピー元パスに「..」は指定できません: %s", file)
        return 1

    source_path = Path(file)

    if source_path.is_dir():
        logger.warning("EFSからS3へのコピーを中止しました。コピー元にディレクトリは指定できません: %s", file)
        return 1

    if not source_path.is_file():
        try:
            # EFSのネガティブキャッシュを更新するため親ディレクトリを読み、再確認する。
            with os.scandir(source_path.parent) as entries:
                for _ in entries:
                    pass
            if not source_path.is_file():
                logger.warning("EFSからS3へのコピーを中止しました。コピー元ファイルが存在しません: %s", file)
                return 1
        except FileNotFoundError:
            logger.warning("EFSからS3へのコピーを中止しました。コピー元ディレクトリが存在しません: %s", source_path.parent)
            return 1

    try:
        # コピー先をS3 URIとして検証する。
        bucket, key = split_s3_uri(
            file_path, "destination", allow_empty_key=True
        )
        if not key:
            key = source_path.name
    except ValueError:
        logger.warning("EFSからS3へのコピーを中止しました。コピー先S3 URIが不正です: %s", file_path)
        return 1

    s3 = create_s3_client()

    # 上書きfalseの場合だけ、S3上に同じキーがないことを事前確認する。
    if not over_write and s3_exists(s3, bucket, key):
        logger.warning("EFSからS3へのコピーを中止しました。コピー先オブジェクトが存在し、上書きが許可されていません: s3://%s/%s", bucket, key)
        return 1

    # NoSuchBucketだけを処理エラーとして扱い、それ以外の転送失敗は呼び出し元へ送出する。
    try:
        s3.upload_file(str(source_path), bucket, key)
    except S3UploadFailedError as exc:
        cause = exc.__context__
        if isinstance(cause, ClientError):
            code = str(cause.response.get("Error", {}).get("Code", ""))
            if code == "NoSuchBucket":
                logger.warning("EFSからS3へのコピーを中止しました。コピー先バケットが存在しません: %s", bucket)
                return 1
        raise
    return 0


def run(argv) -> int:
    """引数を検証し、指定された方向でコピー処理を実行する。

    Args:
        argv: スクリプト名を除く引数。
            [コピー元, コピー先, 上書きオプション（省略可、既定値true）]

    Returns:
        0: コピー成功。
        1: 引数、ファイル、URI、上書きなどの処理エラー。
        -1: S3接続・権限、EFS書込、転送中などの例外。
    """
    try:
        # コピー元・コピー先と、任意の上書きオプションだけを受け付ける。
        if len(argv) < 2 or len(argv) > 3:
            logger.warning("S3/EFSコピーを中止しました。引数は2個または3個指定してください: 指定された引数の数=%d", len(argv))
            return 1

        file = argv[0]
        file_path = argv[1]
        # 上書きオプションの省略時はtrue（上書きする）として扱う。
        over_write_flg = argv[2] if len(argv) > 2 else "true"

        # 空のコピー元またはコピー先は処理できない。
        if not file or not file_path:
            logger.warning("S3/EFSコピーを中止しました。コピー元またはコピー先が空です。")
            return 1

        for path in (file, file_path):
            # s3://または/から始まるパスだけを受け付ける。
            if not (path.startswith(S3_PREFIX) or path.startswith("/")):
                logger.warning("S3/EFSコピーを中止しました。パスにはS3 URIまたは絶対パスを指定してください: %s", path)
                return 1

        # コピー元・コピー先のワイルドカード指定を拒否する。
        if any(char in path for path in (file, file_path) for char in WILDCARD_CHARS):
            logger.warning("S3/EFSコピーを中止しました。ワイルドカードは指定できません: コピー元=%s コピー先=%s", file, file_path)
            return 1

        # 上書き指定をboolに変換し、不正値は引数エラーとして返す。
        try:
            over_write = parse_over_write_flg(over_write_flg)
        except ValueError:
            logger.warning("S3/EFSコピーを中止しました。上書きオプションにはtrueまたはfalseを指定してください: %s", over_write_flg)
            return 1

        # S3 URIの有無で方向を決める。一方だけがS3である必要がある。
        source_is_s3 = file.startswith(S3_PREFIX)
        destination_is_s3 = file_path.startswith(S3_PREFIX)

        if source_is_s3 == destination_is_s3:
            logger.warning("S3/EFSコピーを中止しました。コピー元またはコピー先の一方だけにS3 URIを指定してください: コピー元=%s コピー先=%s", file, file_path)
            return 1

        # 方向ごとの処理に委譲する。
        if source_is_s3:
            result = copy_s3_to_efs(file, file_path, over_write)
            if result == 0:
                logger.info("S3からEFSへのコピーが完了しました: %s -> %s", file, file_path)
            return result
        result = copy_efs_to_s3(file, file_path, over_write)
        if result == 0:
            logger.info("EFSからS3へのコピーが完了しました: %s -> %s", file, file_path)
        return result

    except Exception:
        # 呼び出し元には処理結果コードで例外を通知する。スタックトレースもログに出力する。
        logger.exception("S3/EFSコピー処理中にエラーが発生しました。")
        return -1


def main() -> None:
    """ログをstderrに出力し、処理結果(0、1、-1)をstdoutに出力する。"""
    logging.basicConfig(
        level=logging.INFO,
        stream=sys.stderr,
        format="%(asctime)s %(levelname)s %(name)s %(message)s",
    )
    print(run(sys.argv[1:]))


if __name__ == "__main__":
    main()
