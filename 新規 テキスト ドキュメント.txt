import shutil
from pathlib import Path
from unittest.mock import Mock

import pytest

from kikin.fw.infra.external import fileCopyS3EfsService as service

# テスト用ディレクトリ
TEST_DIR = Path("/home/bttranvu/efs_test/pytest_test")
# テスト用出力ディレクトリ
OUT_DIR = TEST_DIR / "out"


@pytest.fixture(autouse=True)
def setup():
    """各テストの実行前に行う処理。(@BeforeEach相当)"""
    # テスト用ディレクトリが存在していた場合は削除してから作成する
    if TEST_DIR.exists():
        shutil.rmtree(TEST_DIR)
    OUT_DIR.mkdir(parents=True)
    # 実行後は削除しない。(結果を確認できるように残す。Javaと同じ)


def run(*args):
    """処理を実行し、(戻り値, ログ出力のMock)を返す。"""
    copier = service.ApCopyFileS3EfsService.__new__(service.ApCopyFileS3EfsService)
    copier.argv = list(args)
    copier.request = " ".join(args)
    copier.output_log = Mock()
    return copier.do_copy(), copier.output_log


def test1():
    """
    test1：引数が指定されていない場合
    ・引数なし
    """
    # 実行
    result, log = run()

    # 戻り値判定
    assert result == 1

    # ログ判定
    log.assert_called_once()
    assert log.call_args.args[1:] == (
        1, "e.p.9013", ["引数は2個または3個指定してください。", ""])


def test2():
    """
    test2：コピー元ファイルが空文字の場合
    ・コピー元が空文字
    """
    # 入力情報定義
    file_path = str(OUT_DIR / "new.csv")

    # 実行
    result, log = run("", file_path)

    # 戻り値判定
    assert result == 1

    # コピー先にファイルが作成されていないか
    assert not (OUT_DIR / "new.csv").exists(), "コピー先にファイルが作成されない想定"

    # ログ判定
    log.assert_called_once()
    assert log.call_args.args[1:3] == (1, "e.p.9013")
    assert "コピー元またはコピー先が空です。" in log.call_args.args[3][0]
