import os


class FilePath:
    """
    ファイルパス取得クラス
    設定ファイル等のファイルパスを取得する。
    """

    __ap_home_path = os.environ['AP_HOME']
    __etc_apinfra = __ap_home_path + '/etc/apInfra/'
    __temp_apinfra = __ap_home_path + '/temp/apInfra/'
    __catalina_home_path = os.environ.get('CATALINA_HOME')

    @classmethod
    def get_comn_conf_path(cls):
        return cls.__etc_apinfra + 'comnConf.xml'

    @classmethod
    def get_svr_conf_path(cls):
        return cls.__etc_apinfra + 'svrConf.xml'

    @classmethod
    def get_err_msg_path(cls):
        return cls.__etc_apinfra + 'errMsg.xml'

    @classmethod
    def get_queue_conf_path(cls):
        return cls.__etc_apinfra + 'queueConf.xml'

    @classmethod
    def get_lock_file_path(cls):
        return cls.__temp_apinfra + 'lockFile'

    @classmethod
    def get_sys_stat_path(cls):
        return cls.__temp_apinfra + 'SysStat'

    @classmethod
    def get_log_file_path(cls):
        return cls.__ap_home_path + '/log/kikin-monitoring-apbase-pymodule.log'

    @classmethod
    def get_bin_path(cls):
        return cls.__ap_home_path + '/bin'

    @classmethod
    def get_tomcat_path(cls):
        return cls.__catalina_home_path + '/bin'

    @classmethod
    def get_log_path(cls):
        return cls.__ap_home_path + '/log/'

    @classmethod
    def get_backup_dir(cls):
        return cls.__ap_home_path + '/log/backup/'

    @classmethod
    def get_temp_dir(cls):
        return cls.__temp_apinfra + '/apsv/'

    @classmethod
    def get_sqs_temp_dir(cls):
        return cls.__temp_apinfra + '/sqs/'
