from unittest.mock import Mock

from kikin.fw.infra.external import fileCopyS3EfsService as service


def test1(monkeypatch):
    """引数なし：戻り値1とe.p.9013のログ出力を確認する。"""
    copier = service.ApCopyFileS3EfsService.__new__(
        service.ApCopyFileS3EfsService
    )
    copier.argv = []
    copier.request = ""
    output_log = Mock()
    monkeypatch.setattr(copier, "output_log", output_log)

    assert copier.do_copy() == 1
    output_log.assert_called_once()
    assert output_log.call_args.args[1:] == (
        1,
        "e.p.9013",
        ["引数は2個または3個指定してください。", ""],
    )
