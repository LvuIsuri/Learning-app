import uuid
from pathlib import Path

import boto3
import pytest
from botocore.config import Config
from botocore.exceptions import ClientError

from kikin.fw.infra.external import fileCopyS3EfsService as service


S3_TEST_PREFIX = "s3://test-bucket/test/"
S3_ENDPOINT_URL = "http://localhost:4566"
EFS_OUTPUT_DIR = Path.home() / "efs_test" / "out"
TEST_CONTENT = b"pytest Floci S3-EFS integration test\n"
ORIGINAL_CONTENT = b"original content\n"


@pytest.fixture
def floci_s3(monkeypatch):
    monkeypatch.setenv("S3_ENDPOINT_URL", S3_ENDPOINT_URL)
    monkeypatch.setenv("AWS_ACCESS_KEY_ID", "test")
    monkeypatch.setenv("AWS_SECRET_ACCESS_KEY", "test")
    monkeypatch.setenv("AWS_DEFAULT_REGION", "ap-northeast-1")
    monkeypatch.setenv("AWS_EC2_METADATA_DISABLED", "true")
    monkeypatch.delenv("EFS_ALLOWED_DIR", raising=False)

    client = boto3.client(
        "s3",
        endpoint_url=S3_ENDPOINT_URL,
        region_name="ap-northeast-1",
    )
    client.head_bucket(Bucket="test-bucket")
    EFS_OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
    return client


def new_test_object_key() -> str:
    return f"test/pytest-{uuid.uuid4().hex}.txt"


def s3_uri(key: str) -> str:
    return f"{S3_TEST_PREFIX}{Path(key).name}"


def read_s3_object(client, key: str) -> bytes:
    response = client.get_object(Bucket="test-bucket", Key=key)
    try:
        return response["Body"].read()
    finally:
        response["Body"].close()


def assert_s3_object_missing(client, key: str) -> None:
    with pytest.raises(ClientError) as exc_info:
        client.head_object(Bucket="test-bucket", Key=key)
    assert exc_info.value.response["Error"]["Code"] in {
        "404",
        "NoSuchKey",
        "NotFound",
    }


def make_client_error(code: str, operation: str = "HeadObject") -> ClientError:
    return ClientError(
        {"Error": {"Code": code, "Message": "test error"}},
        operation,
    )


def make_source_file() -> Path:
    source = EFS_OUTPUT_DIR / f"pytest-source-{uuid.uuid4().hex}.txt"
    source.write_bytes(TEST_CONTENT)
    return source

def test14(floci_s3, monkeypatch):
    key = new_test_object_key()
    destination = EFS_OUTPUT_DIR / Path(key).name
    floci_s3.put_object(Bucket="test-bucket", Key=key, Body=TEST_CONTENT)
    real_access = service.os.access

    def deny_directory_write(path, mode):
        if Path(path).resolve() == EFS_OUTPUT_DIR.resolve():
            return False
        return real_access(path, mode)

    monkeypatch.setattr(service.os, "access", deny_directory_write)

    try:
        result = service.run([s3_uri(key), str(destination), "false"])

        assert result == -1
        assert not destination.exists()
    finally:
        floci_s3.delete_object(Bucket="test-bucket", Key=key)


def test15(floci_s3, monkeypatch):
    key = new_test_object_key()
    destination = EFS_OUTPUT_DIR / f"pytest-readonly-{uuid.uuid4().hex}.csv"
    destination.write_bytes(ORIGINAL_CONTENT)
    floci_s3.put_object(Bucket="test-bucket", Key=key, Body=TEST_CONTENT)
    real_access = service.os.access

    def deny_destination_write(path, mode):
        if Path(path).resolve() == destination.resolve():
            return False
        return real_access(path, mode)

    monkeypatch.setattr(service.os, "access", deny_destination_write)

    try:
        result = service.run([s3_uri(key), str(destination), "true"])

        assert result == -1
        assert destination.read_bytes() == ORIGINAL_CONTENT
    finally:
        destination.unlink(missing_ok=True)
        floci_s3.delete_object(Bucket="test-bucket", Key=key)


def test16(floci_s3, monkeypatch):

    floci_s3.put_object(Bucket="test-bucket", Key=key, Body=TEST_CONTENT)

    finally:
        destination.unlink(missing_ok=True)
        floci_s3.delete_object(Bucket="test-bucket", Key=key)
