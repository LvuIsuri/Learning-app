import os
import boto3
from botocore.config import Config
from kikin.fw.infra.common import confUtils
from botocore.exceptions import ClientError


class DynamoDBUtils:
    """
    DynamoDBの操作処理を行う。
    """

    GLOBAL_SECONDARY_INDEX01 = "queueMonitorGsi01"

    def __init__(self):
        """
        DynamoDBUtilsのクラスの初期化
        """
        # 設定ファイルからキュー監視テーブルのテーブル名を取得する
        self.__table_name = confUtils.ConfUtils.get_conf_file(confUtils.XmlFiles.QUEUE,
                                                              "queueInfo", "queueMonitorTableName")

        tmp_endpoint = os.environ.get('DYNAMODB_DEBUG_ENDPOINT')
        if tmp_endpoint:
            self.endpoint = tmp_endpoint
        else:
            self.endpoint = "https://dynamodb.ap-northeast-1.amazonaws.com/"
        self.region = "ap-northeast-1"
        # AWS SDK(boto3)の設定(Config)にリトライ回数を5に設定する。
        self.aws_config = Config(
            retries={
                'max_attempts': 5,
            })

    def get_client(self):
        """
        DynamoDBクライアントを生成する。
        :return: DynamoDBクライアント
        """
        return boto3.resource('dynamodb', config=self.aws_config,
                              region_name=self.region,
                              endpoint_url=self.endpoint).Table(self.__table_name)

    @staticmethod
    def put_item(dynamodb_client, item, condition=None):
        """
        DynamoDBに項目を登録する。
        :param dynamodb_client:DynamoDBクライアント
        :param item: 登録する項目
        :param condition: 条件式（オプション）
        :return: 処理結果 True;正常終了
                          False:条件式に一致しないため登録不可
        """
        options = {
            'Item': item,
        }
        if condition is not None:
            options['ConditionExpression'] = condition
        try:
            dynamodb_client.put_item(**options)
        except ClientError as error:
            if error.response['Error']['Code'] == 'ConditionalCheckFailedException':
                return False
            else:
                raise error
        return True

    @staticmethod
    def get_item(dynamodb_client, key, projection, is_consistent_read):
        """
        DynamoDBから一意の項目を取得する。
        :param dynamodb_client: DynamoDBクライアント
        :param key: プライマリキー情報
        :param projection: 取得カラム
        :param is_consistent_read: 強い整合性の有無
        :return:項目情報
        """
        options = {
            'Key': key,
            'ConsistentRead': is_consistent_read,
            'ProjectionExpression': projection,
        }
        response = dynamodb_client.get_item(**options)
        response_item = None
        if 'Item' in response:
            response_item = response['Item']
        return response_item

    @staticmethod
    def delete_item(dynamodb_client, key, condition=None):
        """
        DynamoDBから項目を削除する。
        :param dynamodb_client: DynamoDBクライアント
        :param key: プライマリキー情報
        :param condition: 条件式（オプション）
        :return:処理結果 True;正常終了
                         False:条件式に一致しないため削除不可
        """
        options = {
            'Key': key,
        }
        if condition is not None:
            options['ConditionExpression'] = condition
        try:
            dynamodb_client.delete_item(**options)
        except ClientError as error:
            if error.response['Error']['Code'] == 'ConditionalCheckFailedException':
                return False
            else:
                raise error
        return True

    @staticmethod
    def query_item(dynamodb_client, key_condition, projection, index_name=None, filter_expression=None):
        """
        DynamoDBから指定したパーティションキー、ソートキーに一致する項目情報のリストを取得する。
        :param dynamodb_client: DynamoDBクライアント
        :param key_condition: キー情報
        :param projection: 取得カラム
        :param index_name: インデックス名（オプション）
        :param filter_expression: フィルター式（オプション）
        :return: 項目情報のリスト
        """
        response_list = list()

        options = {
            'KeyConditionExpression': key_condition,
            'ProjectionExpression': projection,
        }
        if index_name is not None:
            options['IndexName'] = index_name
        if filter_expression is not None:
            options['FilterExpression'] = filter_expression

        while True:
            response = dynamodb_client.query(**options)
            response_list += response['Items']

            if "LastEvaluatedKey" in response:
                options['ExclusiveStartKey'] = response['LastEvaluatedKey']
            else:
                break

        return response_list

    @staticmethod
    def query_num_item(dynamodb_client, key_condition, index_name=None, filter_expression=None):
        """
        DynamoDBから指定したパーティションキー、ソートキーに一致する項目情報の件数を取得する。
        :param key_condition: キー情報
        :param index_name: インデックス名（オプション）
        :param filter_expression: フィルター式（オプション）
        :param dynamodb_client: DynamoDBクライアント
        :return: 項目情報の件数
        """
        cnt = 0
        options = {
            'KeyConditionExpression': key_condition,
            'Select': 'COUNT',
        }
        if index_name is not None:
            options['IndexName'] = index_name
        if filter_expression is not None:
            options['FilterExpression'] = filter_expression

        while True:
            response = dynamodb_client.query(**options)
            cnt = cnt + response['Count']

            if "LastEvaluatedKey" in response:
                options['ExclusiveStartKey'] = response['LastEvaluatedKey']
            else:
                break

        return cnt


class QueueMessageInfo:
    """
    キュー監視テーブルに登録するキュー情報モデルを定義する。
    """
    MESSAGE_ID = "id"
    QUEUE_KEY = "queue_key"
    STATUS = "message_status"
    BODY = "message_body"
    HANDLE = "message_handle"
    DATE = "message_date"

    def __init__(self,message_id="", queue_key="", message_status="", message_body="", message_handle="",
                 message_date=""):
        self.id = message_id
        self.queue_key = queue_key
        self.message_status = message_status
        self.message_body = message_body
        self.message_handle = message_handle
        self.message_date = message_date
