def make_client_error(code: str, operation: str = "HeadObject") -> ClientError:
    return ClientError(
        {"Error": {"Code": code, "Message": "test error"}},
        operation,
    )


def write_file(path: Path, content: bytes = TEST_CONTENT) -> Path:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(content)
    return path


def configure_s3_download(client, content: bytes = TEST_CONTENT) -> None:
    client.head_object.return_value = {}
    client.download_file.side_effect = (
        lambda bucket, key, destination: Path(destination).write_bytes(content)
    )
