from pathlib import Path

import pytest
from botocore.exceptions import ClientError

from kikin.fw.infra.external import fileCopyS3EfsService as service


@pytest.mark.parametrize(
    ("uri", "expected"),
    [
        ("s3://test-bucket/sample.csv", ("test-bucket", "sample.csv")),
        (
            "s3://test-bucket/folder/sample.csv",
            ("test-bucket", "folder/sample.csv"),
        ),
    ],
    ids=["正常なS3 URI", "階層を含むS3 URI"],
)
def test_split_s3_uri_success(uri, expected):
    assert service.split_s3_uri(uri) == expected


@pytest.mark.parametrize(
    ("uri", "message"),
    [
        ("file.csv", "コピー元の形式が不正です"),
        ("s3://bucket/*.csv", "コピー元にワイルドカードは指定できません"),
        ("s3://bucket", "コピー元にバケットまたはキーがありません"),
        ("s3://bucket/folder/", "コピー元にディレクトリは指定できません"),
    ],
    ids=[
        "S3 URI形式エラー",
        "ワイルドカード指定エラー",
        "バケットまたはキー不足",
        "ディレクトリ指定エラー",
    ],
)
def test_split_s3_uri_error_message_is_japanese(uri, message):
    with pytest.raises(ValueError, match=message):
        service.split_s3_uri(uri)


@pytest.mark.parametrize(
    ("value", "expected"),
    [
        ("true", True),
        ("TRUE", True),
        ("false", False),
        (" FALSE ", False),
    ],
    ids=["上書きtrue", "大文字TRUE", "上書きfalse", "空白付きfalse"],
)
def test_parse_over_write_flag(value, expected):
    assert service.parse_over_write_flg(value) is expected


def test_parse_over_write_flag_error_message_is_japanese():
    with pytest.raises(
        ValueError,
        match="上書きオプションは true または false を指定してください",
    ):
        service.parse_over_write_flg("yes")


class FakeS3:
    def __init__(self, *, head_error=None):
        self.head_error = head_error
        self.download_calls = []
        self.upload_calls = []

    def head_object(self, **kwargs):
        if self.head_error is not None:
            raise self.head_error

    def download_file(self, bucket, key, destination):
        self.download_calls.append((bucket, key, destination))
        Path(destination).write_bytes(b"sample")

    def upload_file(self, source, bucket, key):
        self.upload_calls.append((source, bucket, key))


def make_client_error(code):
    return ClientError(
        {"Error": {"Code": code, "Message": "test error"}},
        "HeadObject",
    )


@pytest.mark.parametrize(
    ("error_code", "expected"),
    [
        (None, True),
        ("404", False),
        ("NoSuchKey", False),
    ],
    ids=["S3オブジェクト存在", "S3オブジェクトなし404", "S3キーなし"],
)
def test_s3_exists(error_code, expected):
    error = make_client_error(error_code) if error_code else None
    assert service.s3_exists(FakeS3(head_error=error), "bucket", "key") is expected


def test_s3_exists_reraises_forbidden_error():
    s3 = FakeS3(head_error=make_client_error("403"))

    with pytest.raises(ClientError, match="test error"):
        service.s3_exists(s3, "bucket", "key")


@pytest.mark.parametrize(
    ("args", "expected"),
    [
        ([], 1),
        (["source"], 1),
        (["s3://bucket/key", "s3://bucket/other"], 1),
        (["/tmp/source", "/tmp/destination", "yes"], 1),
    ],
    ids=[
        "引数不足",
        "コピー先引数不足",
        "コピー元とコピー先が両方S3",
        "上書きオプション不正",
    ],
)
def test_run_returns_input_error_code(args, expected):
    assert service.run(args) == expected


def test_copy_s3_to_efs_success(monkeypatch, tmp_path):
    s3 = FakeS3()
    monkeypatch.setattr(service, "create_s3_client", lambda: s3)

    result = service.copy_s3_to_efs(
        "s3://test-bucket/sample.csv",
        str(tmp_path),
        True,
    )

    assert result == 0
    assert (tmp_path / "sample.csv").read_bytes() == b"sample"
    assert len(s3.download_calls) == 1


def test_copy_s3_to_efs_returns_error_when_source_is_missing(monkeypatch, tmp_path):
    s3 = FakeS3(head_error=make_client_error("404"))
    monkeypatch.setattr(service, "create_s3_client", lambda: s3)

    result = service.copy_s3_to_efs(
        "s3://test-bucket/missing.csv",
        str(tmp_path),
        True,
    )

    assert result == 1
    assert not (tmp_path / "missing.csv").exists()


def test_run_returns_exception_code_for_s3_forbidden(monkeypatch, tmp_path):
    s3 = FakeS3(head_error=make_client_error("403"))
    monkeypatch.setattr(service, "create_s3_client", lambda: s3)

    result = service.run(
        ["s3://test-bucket/sample.csv", str(tmp_path / "sample.csv"), "true"]
    )

    assert result == -1


def test_copy_efs_to_s3_success(monkeypatch, tmp_path):
    source = tmp_path / "sample.csv"
    source.write_bytes(b"sample")
    s3 = FakeS3()
    monkeypatch.setattr(service, "create_s3_client", lambda: s3)

    result = service.copy_efs_to_s3(
        str(source),
        "s3://test-bucket/sample.csv",
        True,
    )

    assert result == 0
    assert s3.upload_calls == [(str(source), "test-bucket", "sample.csv")]


def test_copy_efs_to_s3_returns_error_when_source_is_missing(monkeypatch, tmp_path):
    monkeypatch.setattr(
        service,
        "create_s3_client",
        lambda: pytest.fail("S3 client must not be created for a missing source"),
    )

    result = service.copy_efs_to_s3(
        str(tmp_path / "missing.csv"),
        "s3://test-bucket/missing.csv",
        True,
    )

    assert result == 1
import uuid
from pathlib import Path

import boto3
import pytest

from kikin.fw.infra.external import fileCopyS3EfsService as service


S3_TEST_PREFIX = "s3://test-bucket/test/"
S3_ENDPOINT_URL = "http://localhost:4566"
EFS_OUTPUT_DIR = Path.home() / "efs_test" / "out"
TEST_CONTENT = b"pytest Floci S3-EFS integration test\n"


@pytest.fixture
def floci_s3(monkeypatch):
    monkeypatch.setenv("S3_ENDPOINT_URL", S3_ENDPOINT_URL)
    monkeypatch.setenv("AWS_ACCESS_KEY_ID", "test")
    monkeypatch.setenv("AWS_SECRET_ACCESS_KEY", "test")
    monkeypatch.setenv("AWS_DEFAULT_REGION", "ap-northeast-1")
    monkeypatch.setenv("AWS_EC2_METADATA_DISABLED", "true")

    client = boto3.client(
        "s3",
        endpoint_url=S3_ENDPOINT_URL,
        region_name="ap-northeast-1",
    )
    client.head_bucket(Bucket="test-bucket")
    EFS_OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
    return client


def new_test_object_key() -> str:
    return f"test/pytest-{uuid.uuid4().hex}.txt"


def test_floci_copies_s3_object_to_efs(floci_s3):
    key = new_test_object_key()
    destination = EFS_OUTPUT_DIR / Path(key).name
    floci_s3.put_object(
        Bucket="test-bucket",
        Key=key,
        Body=TEST_CONTENT,
    )

    try:
        result = service.run(
            [f"{S3_TEST_PREFIX}{Path(key).name}", str(destination), "true"]
        )

        assert result == 0, f"S3からEFSへのコピーが失敗しました: result={result}"
        assert destination.read_bytes() == TEST_CONTENT
    finally:
        destination.unlink(missing_ok=True)
        floci_s3.delete_object(Bucket="test-bucket", Key=key)


def test_floci_copies_efs_file_to_s3(floci_s3):
    key = new_test_object_key()
    source = EFS_OUTPUT_DIR / f"pytest-source-{uuid.uuid4().hex}.txt"
    source.write_bytes(TEST_CONTENT)

    try:
        result = service.run(
            [str(source), f"{S3_TEST_PREFIX}{Path(key).name}", "true"]
        )

        assert result == 0, f"EFSからS3へのコピーが失敗しました: result={result}"
        downloaded = floci_s3.get_object(Bucket="test-bucket", Key=key)["Body"].read()
        assert downloaded == TEST_CONTENT
    finally:
        source.unlink(missing_ok=True)
        floci_s3.delete_object(Bucket="test-bucket", Key=key)
